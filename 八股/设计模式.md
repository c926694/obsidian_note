# option
- 定义一个可选参数结构体
- 定义修改可选参数对象属性的函数类型
- New函数传入可变修改函数
- with用闭包实现返回函数
```go
package main  
  
import "fmt"  
  
type UserOptions struct {  
    Name string  
    Age  int  
}  
  
type UserOption func(*UserOptions)  
  
type User struct {  
    *UserOptions  
    Sex string  
}  
  
func NewUser(sex string, userOpts ...UserOption) *User {  
    opts := &UserOptions{  
       Name: "张三",  
       Age:  18,  
    }  
  
    for _, opt := range userOpts {  
       opt(opts)  
    }  
    return &User{  
       UserOptions: opts,  
       Sex:         sex,  
    }  
}  
  
func WithName(name string) UserOption {  
    return func(o *UserOptions) {  
       o.Name = name  
    }  
}  
  
func WithAge(age int) UserOption {  
    return func(o *UserOptions) {  
       o.Age = age  
    }  
}  
func main() {  
    u := NewUser("男", WithName("李四"))  
    fmt.Printf("%#v\n", u)  
}
```
